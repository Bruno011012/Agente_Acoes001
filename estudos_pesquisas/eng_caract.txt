import pandas as pd
import numpy as np
from scipy import signal
from scipy.signal import (
    detrend, savgol_filter, hilbert, find_peaks,
    butter, sosfiltfilt, welch
)

def engenharia_features_signal(df: pd.DataFrame) -> pd.DataFrame:
    """
    Engenharia de características usando scipy.signal
    Alvo: Close
    Features base: Open, High, Low, Volume, Date, Mes, Ano
    """
    
    df = df.copy()
    df = df.sort_values("Date").reset_index(drop=True)
    
    # ============================================================
    # 1. Features básicas de preço
    # ============================================================
    df["Range"] = df["High"] - df["Low"]                    # Amplitude do dia
    df["Body"] = df["Close"] - df["Open"]                   # Corpo do candle
    df["Upper_Shadow"] = df["High"] - df[["Open", "Close"]].max(axis=1)
    df["Lower_Shadow"] = df[["Open", "Close"]].min(axis=1) - df["Low"]
    df["Return"] = df["Close"].pct_change()                 # Retorno simples
    df["Log_Return"] = np.log(df["Close"] / df["Close"].shift(1))
    
    # ============================================================
    # 2. DETREND (remove tendência)
    # ============================================================
    df["Close_Detrend"] = detrend(df["Close"].fillna(method="ffill"), type="linear")
    df["Volume_Detrend"] = detrend(df["Volume"].fillna(method="ffill"), type="linear")
    
    # ============================================================
    # 3. SAVITZKY-GOLAY (suavização preservando picos)
    # ============================================================
    window = 11 if len(df) > 11 else 5
    if window % 2 == 0:
        window += 1
        
    df["Close_SavGol"] = savgol_filter(df["Close"].fillna(method="ffill"), 
                                       window_length=window, polyorder=3)
    df["Volume_SavGol"] = savgol_filter(df["Volume"].fillna(method="ffill"), 
                                        window_length=window, polyorder=2)
    
    # ============================================================
    # 4. FILTROS Butterworth
    # ============================================================
    # Frequência de amostragem aproximada (1 por dia)
    fs = 1.0  
    
    # Low-pass (tendência de médio prazo)
    sos_lp = butter(3, 0.1, btype="low", fs=fs, output="sos")
    df["Close_LowPass"] = sosfiltfilt(sos_lp, df["Close"].fillna(method="ffill"))
    
    # High-pass (ruído / movimentos de curto prazo)
    sos_hp = butter(3, 0.2, btype="high", fs=fs, output="sos")
    df["Close_HighPass"] = sosfiltfilt(sos_hp, df["Close"].fillna(method="ffill"))
    
    # ============================================================
    # 5. HILBERT TRANSFORM (envelope e fase)
    # ============================================================
    analytic = hilbert(df["Close_Detrend"].fillna(0))
    df["Envelope"] = np.abs(analytic)
    df["Inst_Phase"] = np.unwrap(np.angle(analytic))
    df["Inst_Freq"] = np.append([0], np.diff(df["Inst_Phase"]) / (2.0 * np.pi))
    
    # ============================================================
    # 6. DETECÇÃO DE PICOS E VALES
    # ============================================================
    # Picos (máximas locais)
    peaks, _ = find_peaks(df["Close"].fillna(method="ffill"), distance=5, prominence=0.5)
    df["Is_Peak"] = 0
    df.loc[peaks, "Is_Peak"] = 1
    
    # Vales (mínimas locais)
    valleys, _ = find_peaks(-df["Close"].fillna(method="ffill"), distance=5, prominence=0.5)
    df["Is_Valley"] = 0
    df.loc[valleys, "Is_Valley"] = 1
    
    # Distância desde o último pico/vale
    df["Dist_Last_Peak"] = df["Is_Peak"].iloc[::-1].cumsum().iloc[::-1]
    df["Dist_Last_Valley"] = df["Is_Valley"].iloc[::-1].cumsum().iloc[::-1]
    
    # ============================================================
    # 7. FEATURES DE VOLUME + PREÇO
    # ============================================================
    df["Volume_Change"] = df["Volume"].pct_change()
    df["Price_Volume"] = df["Close"] * df["Volume"]
    df["VWAP_proxy"] = (df["High"] + df["Low"] + df["Close"]) / 3 * df["Volume"]
    
    # ============================================================
    # 8. LAGS e MÉDIAS MÓVEIS (clássicas, mas úteis)
    # ============================================================
    for lag in [1, 2, 3, 5, 10]:
        df[f"Close_lag_{lag}"] = df["Close"].shift(lag)
        df[f"Volume_lag_{lag}"] = df["Volume"].shift(lag)
        df[f"Return_lag_{lag}"] = df["Return"].shift(lag)
    
    df["MA_5"] = df["Close"].rolling(5).mean()
    df["MA_10"] = df["Close"].rolling(10).mean()
    df["MA_20"] = df["Close"].rolling(20).mean()
    df["Volatility_10"] = df["Return"].rolling(10).std()
    
    # ============================================================
    # 9. FEATURES TEMPORAIS (já existentes + derivadas)
    # ============================================================
    # Assume que já existem: Date, Mes, Ano
    df["Dia_Semana"] = pd.to_datetime(df["Date"]).dt.dayofweek
    df["Dia_Mes"] = pd.to_datetime(df["Date"]).dt.day
    df["Semana_Ano"] = pd.to_datetime(df["Date"]).dt.isocalendar().week.astype(int)
    
    # ============================================================
    # 10. ENERGIA ESPECTRAL (Welch) em janelas
    # ============================================================
    def spectral_energy(series, fs=1.0):
        if len(series) < 16:
            return 0.0
        f, Pxx = welch(series, fs=fs, nperseg=min(16, len(series)))
        return np.sum(Pxx)
    
    df["Spectral_Energy"] = df["Close"].rolling(20).apply(
        lambda x: spectral_energy(x.values), raw=False
    )
    
    # ============================================================
    # Limpeza final
    # ============================================================
    df = df.replace([np.inf, -np.inf], np.nan)
    
    return df


# ============================================================

# ============================================================
if __name__ == "__main__":
    # Simula uma base
    np.random.seed(42)
    n = 200
    dates = pd.date_range("2025-01-01", periods=n, freq="B")
    
    df = pd.DataFrame({
        "Date": dates,
        "Open": np.random.uniform(100, 150, n),
        "High": np.random.uniform(150, 180, n),
        "Low": np.random.uniform(90, 120, n),
        "Close": np.random.uniform(100, 160, n),
        "Volume": np.random.randint(1_000_000, 10_000_000, n),
        "Mes": dates.month,
        "Ano": dates.year
    })
    
    # Garante High > Low e coerência básica
    df["High"] = df[["Open", "Close", "High"]].max(axis=1)
    df["Low"] = df[["Open", "Close", "Low"]].min(axis=1)
    
    # Aplica a engenharia
    df_features = engenharia_features_signal(df)
    
    print("Shape original:", df.shape)
    print("Shape com features:", df_features.shape)
    print("\nNovas colunas criadas:")
    print([c for c in df_features.columns if c not in df.columns])
    
    print("\nAmostra das features:")
    print(df_features[["Close", "Close_Detrend", "Close_SavGol", "Envelope", 
                       "Is_Peak", "MA_10", "Spectral_Energy"]].tail(10))
